# plugin-visualizer

```js
import { visualizer } from "rollup-plugin-visualizer";

module.exports = {
  plugins: [
    // put it last
    visualizer(),
  ],
};


const config = {
  plugins: [
    visualizer({
      emitFile: true,
      filename: "stats.html",
    }),
  ],
};
```
